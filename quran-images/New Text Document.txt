    <script type="text/service-worker" id="sw">
        const CACHE_NAME = 'quran-reader-cache-v1';
        const APP_SHELL_URL = '/';

        self.addEventListener('install', event => {
            event.waitUntil(
                caches.open(CACHE_NAME)
                    .then(cache => {
                        // Cache the main HTML file itself. The path '/' may need adjustment
                        // depending on deployment, but for a single file app at the root, it's correct.
                        // We fetch the app shell with a cache-busting query to ensure we get the latest version from the network.
                        return cache.add(new Request(APP_SHELL_URL, {cache: 'reload'}));
                    })
            );
            self.skipWaiting();
        });

        self.addEventListener('activate', event => {
            event.waitUntil(
                caches.keys().then(cacheNames => {
                    return Promise.all(
                        cacheNames.filter(name => name !== CACHE_NAME).map(name => caches.delete(name))
                    );
                })
            );
            self.clients.claim();
        });

        self.addEventListener('fetch', event => {
            // We only handle navigation requests for the app shell.
            // All other assets (images, data) are served from IndexedDB, not the network.
            if (event.request.mode === 'navigate') {
                event.respondWith(
                    caches.match(APP_SHELL_URL)
                        .then(response => {
                            return response || fetch(event.request);
                        })
                );
            }
        });
    </script>

        // --- DATABASE HELPERS ---
        function openDB() {
            return new Promise((resolve, reject) => {
                const request = indexedDB.open(DB_NAME, DB_VERSION);

                request.onupgradeneeded = event => {
                    const dbInstance = event.target.result;
                    if (!dbInstance.objectStoreNames.contains(IMAGE_STORE_NAME)) {
                        dbInstance.createObjectStore(IMAGE_STORE_NAME); // Key is page number
                    }
                    if (!dbInstance.objectStoreNames.contains(DATA_STORE_NAME)) {
                        dbInstance.createObjectStore(DATA_STORE_NAME); // Key is data type ('lastPage', 'bookmarks', etc.)
                    }
                };

                request.onsuccess = event => {
                    db = event.target.result;
                    resolve(db);
                };

                request.onerror = event => {
                    console.error('Database error:', event.target.error);
                    reject('Database error: ' + event.target.error);
                };
            });
        }

        function dbGet(storeName, key) {
            return new Promise((resolve, reject) => {
                if (!db) { reject("DB not initialized"); return; }
                const transaction = db.transaction(storeName, 'readonly');
                const store = transaction.objectStore(storeName);
                const request = store.get(key);
                request.onsuccess = () => resolve(request.result);
                request.onerror = () => reject(request.error);
            });
        }

        function dbSet(storeName, key, value) {
            return new Promise((resolve, reject) => {
                if (!db) { reject("DB not initialized"); return; }
                const transaction = db.transaction(storeName, 'readwrite');
                const store = transaction.objectStore(storeName);
                const request = store.put(value, key);
                request.onsuccess = () => resolve();
                request.onerror = () => reject(request.error);
            });
        }
        
        function dbDelete(storeName, key) {
            return new Promise((resolve, reject) => {
                if (!db) { reject("DB not initialized"); return; }
                const transaction = db.transaction(storeName, 'readwrite');
                const store = transaction.objectStore(storeName);
                const request = store.delete(key);
                request.onsuccess = () => resolve();
                request.onerror = () => reject(request.error);
            });
        }

        function checkImagesExist() {
            return new Promise(async (resolve) => {
                try {
                    const image = await dbGet(IMAGE_STORE_NAME, 1);
                    resolve(!!image);
                } catch (e) {
                    resolve(false);
                }
            });
        }
        async function autoImportImages() {
            for (let i = 1; i <= TOTAL_PAGES; i++) {
                try {
                    // Fetch the image assuming it's in the same folder
                    const response = await fetch(`${i}.png`);
                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }
                    const imageBlob = await response.blob();
                    
                    // Store the blob in IndexedDB
                    await dbSet(IMAGE_STORE_NAME, i, imageBlob);

                    // Update progress UI
                    const progress = ((i / TOTAL_PAGES) * 100).toFixed(1);
                    importProgressBar.style.width = `${progress}%`;
                    importStatus.textContent = `Loading page ${i} / ${TOTAL_PAGES}...`;

                } catch (error) {
                    console.error(`Failed to load and store page ${i}.png:`, error);
                    importStatus.innerHTML = `<b>Error:</b> Could not load page <b>${i}.png</b>. <br><br> Please ensure all 604 image files are in the same directory as this HTML file and that you are running it from a web server, not as a local file. Then, refresh the page.`;
                    // Stop the import process on the first error
                    return;
                }
            }

            importStatus.textContent = 'Setup complete! Initializing app...';
            // Set sample data on first install
            await dbSet(DATA_STORE_NAME, 'bookmarks', [1, 78, 255, 604]);
            setTimeout(() => window.location.reload(), 1500);
        }
        // --- SETUP SCREEN LOGIC ---
        function initSetup() {
            appLoader.classList.add('hidden');
            setupScreen.classList.remove('hidden');
            importProgressContainer.classList.remove('hidden');
            autoImportImages(); // Start the automatic import

            handleFiles(e.target.files);
        }

        async function handleFiles(files) {
            if (files.length !== TOTAL_PAGES) {
                alert(`Please select or drop exactly ${TOTAL_PAGES} image files.`);
                return;
            }

            importProgressContainer.classList.remove('hidden');
            dropZone.classList.add('hidden');
            
            const imageFiles = Array.from(files).sort((a, b) => {
                const numA = parseInt(a.name.match(/\d+/)[0], 10);
                const numB = parseInt(b.name.match(/\d+/)[0], 10);
                return numA - numB;
            });

            for (let i = 0; i < imageFiles.length; i++) {
                const file = imageFiles[i];
                const pageNum = i + 1;
                
                try {
                    await dbSet(IMAGE_STORE_NAME, pageNum, file);
                    const progress = ((pageNum / TOTAL_PAGES) * 100).toFixed(1);
                    importProgressBar.style.width = `${progress}%`;
                    importStatus.textContent = `Importing page ${pageNum} / ${TOTAL_PAGES}...`;
                } catch (error) {
                    console.error(`Failed to import page ${pageNum}:`, error);
                    importStatus.textContent = `Error importing page ${pageNum}. Please refresh and try again.`;
                    return;
                }
            }

            importStatus.textContent = 'Import complete! Initializing app...';
            // Set some sample data on first install
            await dbSet(DATA_STORE_NAME, 'bookmarks', [1, 78, 255, 604]);
            setTimeout(() => window.location.reload(), 1500);
        }

        // --- READER SCREEN LOGIC ---
        async function initReader() {
            // Load user data
            const [lastReadPage, savedBookmarks, savedTheme] = await Promise.all([
                dbGet(DATA_STORE_NAME, 'lastReadPage'),
                dbGet(DATA_STORE_NAME, 'bookmarks'),
                dbGet(DATA_STORE_NAME, 'isDarkMode')
            ]);
            
            currentPage = lastReadPage || 1;
            bookmarks = savedBookmarks || [1, 78, 255, 604]; // Add sample data if none exists
            isDarkMode = savedTheme || false;
            
            setupEventListeners();
            applyTheme();
            updateBookmarksUI();
            
            await loadPage(currentPage, true);

            appLoader.classList.add('hidden');
            readerScreen.classList.remove('hidden');
        }

        function setupEventListeners() {
            // Navigation
            prevBtn.addEventListener('click', () => changePage(currentPage - 1));
            nextBtn.addEventListener('click', () => changePage(currentPage + 1));
            pageSlider.addEventListener('input', () => pageNumInput.value = pageSlider.value);
            pageSlider.addEventListener('change', () => changePage(parseInt(pageSlider.value, 10)));
            pageNumInput.addEventListener('change', () => changePage(parseInt(pageNumInput.value, 10)));
            document.addEventListener('keydown', handleKeydown);

            // UI Controls
            readerScreen.addEventListener('click', handleScreenTap);
            menuBtn.addEventListener('click', toggleMenu);
            
            // Bookmarks
            bookmarkBtn.addEventListener('click', toggleBookmark);

            // Theme
            themeSwitch.addEventListener('change', toggleTheme);
            
            // Backup & Restore
            exportBtn.addEventListener('click', exportData);
            importInput.addEventListener('change', importData);

            // Zoom & Pan
            imageContainer.addEventListener('touchstart', handleTouchStart);
            imageContainer.addEventListener('touchmove', handleTouchMove);
            imageContainer.addEventListener('touchend', handleTouchEnd);
        }
        
        let preloadedPages = {};
        async function preloadPage(pageNum) {
            if (pageNum < 1 || pageNum > TOTAL_PAGES || preloadedPages[pageNum]) {
                return;
            }
            try {
                const blob = await dbGet(IMAGE_STORE_NAME, pageNum);
                if (blob) {
                    preloadedPages[pageNum] = URL.createObjectURL(blob);
                }
            } catch(e) {
                console.warn("Could not preload page", pageNum, e);
            }
        }
        
        async function loadPage(pageNum, isInitialLoad = false) {
            pageNum = Math.max(1, Math.min(TOTAL_PAGES, pageNum));

            if (!isInitialLoad) {
                quranPageImg.style.opacity = '0';
            }
            
            resetZoom();

            try {
                let objectURL;
                if (preloadedPages[pageNum]) {
                    objectURL = preloadedPages[pageNum];
                    delete preloadedPages[pageNum]; // Use it, then remove from preload cache
                } else {
                    const blob = await dbGet(IMAGE_STORE_NAME, pageNum);
                    if (!blob) throw new Error(`Image for page ${pageNum} not found.`);
                    objectURL = URL.createObjectURL(blob);
                }
                
                // Revoke old URL to free memory, except for preloaded ones
                if (quranPageImg.src && !Object.values(preloadedPages).includes(quranPageImg.src)) {
                     URL.revokeObjectURL(quranPageImg.src);
                }

                quranPageImg.src = objectURL;
                quranPageImg.onload = () => {
                    quranPageImg.style.opacity = '1';
                };

                currentPage = pageNum;
                updateUIAfterPageChange();
                dbSet(DATA_STORE_NAME, 'lastReadPage', currentPage);

                // Preload adjacent pages
                preloadPage(currentPage - 1);
                preloadPage(currentPage + 1);
                preloadPage(currentPage - 2);
                preloadPage(currentPage + 2);

            } catch (error) {
                console.error('Error loading page:', error);
                alert('Could not load page. The database might be corrupted.');
            }
        }
        
        function changePage(pageNum) {
            if (pageNum >= 1 && pageNum <= TOTAL_PAGES) {
                loadPage(pageNum);
            }
        }
        
        function updateUIAfterPageChange() {
            pageInfo.textContent = `Page ${currentPage} / ${TOTAL_PAGES}`;
            pageSlider.value = currentPage;
            pageNumInput.value = currentPage;
            
            // Update bookmark status
            if (bookmarks.includes(currentPage)) {
                bookmarkBtn.classList.add('bookmarked');
            } else {
                bookmarkBtn.classList.remove('bookmarked');
            }
        }
        
        function handleKeydown(e) {
            if (document.activeElement === pageNumInput) return;
            if (e.key === 'ArrowRight') nextBtn.click();
            if (e.key === 'ArrowLeft') prevBtn.click();
        }

        // --- UI VISIBILITY ---
        function handleScreenTap(e) {
            // If tap is on a button or inside the menu, do nothing
            if (e.target.closest('button, input, #menu-panel')) {
                return;
            }
            
            // If menu is open, close it
            if (menuOpen) {
                toggleMenu();
                return;
            }

            // Toggle controls
            controlsVisible = !controlsVisible;
            readerScreen.classList.toggle('controls-visible', controlsVisible);
            
            // On mobile, show nav buttons too
            if ('ontouchstart' in window) {
                document.querySelector('.ui-controls').classList.toggle('touch-active', controlsVisible);
            }

            clearTimeout(controlsTimeout);
            if (controlsVisible) {
                controlsTimeout = setTimeout(() => {
                    controlsVisible = false;
                    readerScreen.classList.remove('controls-visible');
                    document.querySelector('.ui-controls').classList.remove('touch-active');
                }, 4000);
            }
        }

        // --- MENU & BOOKMARKS ---
        function toggleMenu(e) {
            if (e) e.stopPropagation();
            menuOpen = !menuOpen;
            readerScreen.classList.toggle('menu-open', menuOpen);
        }

        function toggleBookmark() {
            const index = bookmarks.indexOf(currentPage);
            if (index > -1) {
                bookmarks.splice(index, 1);
            } else {
                bookmarks.push(currentPage);
                bookmarks.sort((a,b) => a - b);
            }
            dbSet(DATA_STORE_NAME, 'bookmarks', bookmarks);
            updateBookmarksUI();
            updateUIAfterPageChange(); // to update the star icon color
        }
        
        function updateBookmarksUI() {
            bookmarksList.innerHTML = '';
            if (bookmarks.length === 0) {
                bookmarksList.innerHTML = '<li class="no-bookmarks">No bookmarks yet.</li>';
                return;
            }
            
            bookmarks.forEach(pageNum => {
                const li = document.createElement('li');
                li.dataset.page = pageNum;
                li.innerHTML = `<span>Page ${pageNum}</span><button class="delete-bookmark-btn" data-page="${pageNum}">×</button>`;
                
                li.addEventListener('click', (e) => {
                    if (e.target.classList.contains('delete-bookmark-btn')) return;
                    changePage(pageNum);
                    toggleMenu();
                });
                
                li.querySelector('.delete-bookmark-btn').addEventListener('click', (e) => {
                    e.stopPropagation();
                    const pageToDelete = parseInt(e.target.dataset.page, 10);
                    bookmarks = bookmarks.filter(p => p !== pageToDelete);
                    dbSet(DATA_STORE_NAME, 'bookmarks', bookmarks);
                    updateBookmarksUI();
                    updateUIAfterPageChange();
                });
                
                bookmarksList.appendChild(li);
            });
        }
        
        // --- THEME ---
        function applyTheme() {
            document.documentElement.classList.toggle('dark-mode', isDarkMode);
            themeSwitch.checked = isDarkMode;
            // Update theme-color meta tag for PWA
            document.querySelector('meta[name="theme-color"]').setAttribute('content', isDarkMode ? '#1a1a1a' : '#f4f4f4');
        }

        function toggleTheme() {
            isDarkMode = !isDarkMode;
            dbSet(DATA_STORE_NAME, 'isDarkMode', isDarkMode);
            applyTheme();
        }

        // --- ZOOM & PAN ---
        function resetZoom() {
            zoomState = { scale: 1, x: 0, y: 0 };
            applyTransform();
        }

        function applyTransform() {
            // Clamp values to prevent excessive scaling or panning off-screen
            zoomState.scale = Math.max(1, Math.min(zoomState.scale, 4));

            const maxPanX = (quranPageImg.offsetWidth * zoomState.scale - imageContainer.clientWidth) / 2;
            const maxPanY = (quranPageImg.offsetHeight * zoomState.scale - imageContainer.clientHeight) / 2;
            
            zoomState.x = Math.max(-maxPanX, Math.min(zoomState.x, maxPanX));
            zoomState.y = Math.max(-maxPanY, Math.min(zoomState.y, maxPanY));
            
            // If scale is 1, reset pan
            if(zoomState.scale === 1) {
                zoomState.x = 0;
                zoomState.y = 0;
            }

            quranPageImg.style.transform = `translate(${zoomState.x}px, ${zoomState.y}px) scale(${zoomState.scale})`;
        }

        function getDistance(p1, p2) {
            return Math.sqrt(Math.pow(p2.clientX - p1.clientX, 2) + Math.pow(p2.clientY - p1.clientY, 2));
        }

        function handleTouchStart(e) {
            e.preventDefault();

            // Double tap to zoom
            const now = new Date().getTime();
            if (now - lastTap < 300) {
                if (zoomState.scale > 1) {
                    resetZoom();
                } else {
                    zoomState.scale = 2.5;
                    applyTransform();
                }
                lastTap = 0;
                return;
            }
            lastTap = now;

            if (e.touches.length === 1) {
                zoomState.startX = e.touches[0].clientX - zoomState.x;
                zoomState.startY = e.touches[0].clientY - zoomState.y;
            }
            if (e.touches.length === 2) {
                zoomState.initialDistance = getDistance(e.touches[0], e.touches[1]);
            }
        }

        function handleTouchMove(e) {
            e.preventDefault();
            if (zoomState.scale <= 1 && e.touches.length === 1) return; // Don't pan if not zoomed

            if (e.touches.length === 1) {
                zoomState.x = e.touches[0].clientX - zoomState.startX;
                zoomState.y = e.touches[0].clientY - zoomState.startY;
            }
            if (e.touches.length === 2) {
                const newDist = getDistance(e.touches[0], e.touches[1]);
                zoomState.scale = zoomState.scale * (newDist / zoomState.initialDistance);
                zoomState.initialDistance = newDist;
            }
            applyTransform();
        }

        function handleTouchEnd(e) {
            e.preventDefault();
            // Final adjustments or snapping can be done here if needed
        }
        
        // --- BACKUP & RESTORE ---
        async function exportData() {
            try {
                const dataToExport = {
                    lastReadPage: await dbGet(DATA_STORE_NAME, 'lastReadPage') || 1,
                    bookmarks: await dbGet(DATA_STORE_NAME, 'bookmarks') || [],
                    isDarkMode: await dbGet(DATA_STORE_NAME, 'isDarkMode') || false,
                };
                
                const jsonString = JSON.stringify(dataToExport, null, 2);
                const blob = new Blob([jsonString], { type: 'application/json' });
                const url = URL.createObjectURL(blob);
                
                const a = document.createElement('a');
                a.href = url;
                a.download = `quran-reader-backup-${new Date().toISOString().split('T')[0]}.json`;
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
                URL.revokeObjectURL(url);
            } catch (error) {
                console.error('Export failed:', error);
                alert('Failed to export data.');
            }
        }

        function importData(e) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = async (event) => {
                try {
                    const data = JSON.parse(event.target.result);
                    if (!data.hasOwnProperty('lastReadPage') || !data.hasOwnProperty('bookmarks')) {
                        throw new Error('Invalid backup file format.');
                    }
                    
                    if (confirm('This will overwrite your current progress and bookmarks. Are you sure?')) {
                        await Promise.all([
                            dbSet(DATA_STORE_NAME, 'lastReadPage', data.lastReadPage),
                            dbSet(DATA_STORE_NAME, 'bookmarks', data.bookmarks),
                            dbSet(DATA_STORE_NAME, 'isDarkMode', data.isDarkMode || false)
                        ]);
                        alert('Data imported successfully! The app will now reload.');
                        window.location.reload();
                    }
                } catch (error) {
                    console.error('Import failed:', error);
                    alert('Import failed. The file may be corrupt or in the wrong format.');
                } finally {
                    // Reset file input to allow re-selection of the same file
                    importInput.value = '';
                }
            };
            reader.readAsText(file);
        }

        // --- PWA SERVICE WORKER REGISTRATION ---
        function registerServiceWorker() {
            if ('serviceWorker' in navigator) {
                const swScriptElement = document.getElementById('sw');
                if (!swScriptElement) {
                    console.error("Service worker script element not found.");
                    return;
                }
                const swScript = swScriptElement.textContent;
                const swBlob = new Blob([swScript], { type: 'application/javascript' });
                const swUrl = URL.createObjectURL(swBlob);

                navigator.serviceWorker.register(swUrl)
                    .then(registration => {
                        console.log('Service Worker registered with scope:', registration.scope);
                    })
                    .catch(error => {
                        console.error('Service Worker registration failed:', error);
                    });
            }
        }


        // --- INITIALIZATION ---
        async function main() {
            try {
                await openDB();
                const imagesExist = await checkImagesExist();
                if (imagesExist) {
                    initReader();
                } else {
                    initSetup();
                }
                registerServiceWorker();
            } catch (error) {
                console.error('App initialization failed:', error);
                appLoader.innerHTML = `<p style="color:red; text-align:center;">A critical error occurred. Please clear your browser cache/data for this site and try again. <br><br> Error: ${error}</p>`;
            }
        }

        main();
    });
